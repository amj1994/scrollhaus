CORTEXA — complete build spec

A finished, self-contained single-file build supplied ready-made: a
neurotech/AI hero built around a scroll-scrubbed macro neuron-network video
plate (glowing dendrites against black, genuinely dense high-motion footage),
a scramble-text headline effect, a cinematic mid-page pull-quote section, and
a Swiper coverflow stat-card carousel.

Video: originally hosted on Gumlet at raw capture bitrate (31.9Mbps, 1080p,
with an unused silent audio track) — extremely dense, high-frequency footage
(fine particle sparkle against pure black, the hardest case for H.264) that
resists compression harder than a typical hero clip. Stripped audio,
re-encoded with a dense forced keyframe interval (crf 22, -g 6 -keyint_min 6
-sc_threshold 0 — a keyframe roughly every 0.25s) and self-hosted at
assets/video/hero.mp4 instead. The dense keyframe interval matters here more
than almost anywhere else in this library: the video is scrubbed by setting
`video.currentTime` directly from scroll position (no WebCodecs), so a sparse
keyframe interval would make scrubbing visibly step between frames instead of
interpolating smoothly — a real bug found and fixed elsewhere in this library
this same session.

Motion: Lenis smooth scroll wired in (the build's own choice, `@studio-freight/lenis`
from unpkg, loaded conditionally on desktop only — window.__lenis exposed for
the capture-script convention).

Stack: plain static HTML/CSS/vanilla JS. No build step, no framework.

## Setup
Create one file named index.html with exactly this content and serve it as a
static file (any static server, e.g. `python3 -m http.server`). Needs
internet — Swiper, Bootstrap Icons, and Google Fonts load from CDN. The hero
video must be present at `assets/video/hero.mp4` relative to the served root.

<script>
// ── Constants ────────────────────────────────────────────────────────────────
const VIDEO_URL = 'assets/video/hero.mp4';

// ── State ────────────────────────────────────────────────────────────────────
let scrollProgress = 0, smoothScrollProgress = 0;
let phase = 'loading';
let entranceStart = null;
const ENTRANCE_DURATION = 1400;

// ── Helpers ──────────────────────────────────────────────────────────────────
function clamp01(v){return Math.max(0,Math.min(1,v))}
function ease(t){return 1-Math.pow(1-t,3)}

// ── Video setup ──────────────────────────────────────────────────────────────
const video = document.getElementById('hero-video');
video.autoplay = false;
let videoReady = false, isSeeking = false, nextSeekTime = null;
video.addEventListener('loadedmetadata', function(){video.pause(); videoReady=true; if(video.readyState>=3)tryStartEntrance();});
video.addEventListener('canplay', function(){if(video.readyState>=3)tryStartEntrance();});
video.addEventListener('seeked', function(){isSeeking=false; if(nextSeekTime!=null){var t=nextSeekTime;nextSeekTime=null;seekTo(t);}});
function seekTo(t){if(isSeeking){nextSeekTime=t;return;} isSeeking=true; video.currentTime=t;}

// ── Entrance ─────────────────────────────────────────────────────────────────
function tryStartEntrance(){
  if(phase!=='loading')return;
  phase='animating';
  entranceStart=performance.now();
}
// Safety timeout at 2000ms
setTimeout(function(){tryStartEntrance();}, 2000);

// ── ScrambleIn ───────────────────────────────────────────────────────────────
var GLYPHS='ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789!@#$%^&*()_+~|}{[]:;?><';
function rg(){return GLYPHS[Math.floor(Math.random()*GLYPHS.length)];}
var scrambleEls=[];
document.querySelectorAll('[data-scramble-in]').forEach(function(el){
  var text=el.dataset.text||el.textContent;
  var delay=parseInt(el.dataset.delay||'0');
  el.dataset.text=text;
  el.textContent=' '.repeat(text.length);
  scrambleEls.push({el:el,text:text,delay:delay,state:'hidden',animStart:null,animDir:null,animDuration:0});
});
function triggerScramble(s,dir){
  if(s.animDir===dir&&s.animStart!=null)return;
  s.animDir=dir;
  s.animDuration=dir==='in'?900:700;
  s.animStart=performance.now();
  s.state='animating';
}
function animateScramble(ts){
  scrambleEls.forEach(function(s){
    if(!s.animStart)return;
    var elapsed=ts-s.animStart;
    var prog=Math.min(1,elapsed/s.animDuration);
    var rev=s.animDir==='in'?prog:1-prog;
    var chars=s.text.split('').map(function(ch,i){
      if(ch===' ')return ' ';
      var threshold=i/s.text.length;
      if(rev>threshold)return ch;
      return rg();
    });
    s.el.textContent=chars.join('');
    if(prog>=1){s.animStart=null;s.state=s.animDir==='in'?'visible':'hidden';}
  });
}

// ── Stats data ───────────────────────────────────────────────────────────────
var statsData=[
  {title:'Neural Latency',value:'0.4ms',footer:'Signal processing',detail:'End-to-end synaptic signal capture and routing latency measured across 1,024 concurrent electrode channels.'},
  {title:'Pathway Coverage',value:'98.7%',footer:'Cortical mapping',detail:'Percentage of primary motor and sensory cortex regions successfully mapped during baseline calibration sessions.'},
  {title:'Cognitive Channels',value:'2048',footer:'Active streams',detail:'Simultaneous cognitive data streams processed per session across distributed neural pathway monitors.'},

  {title:'Load Reduction',value:'73%',footer:'Cognitive overhead',detail:'Average reduction in reported cognitive load among users after four weeks of adaptive intelligence calibration.'},
];
var wrapper=document.getElementById('stats-wrapper');

  wrapper.appendChild(slide);
});

var statsRevealed=false;
function maybeRevealStats(){
  if(statsRevealed)return;
  var sect=document.getElementById('stats-section');
  if(sect.getBoundingClientRect().top<window.innerHeight*0.9){
    statsRevealed=true;
    sect.querySelectorAll('.stat-card').forEach(function(c,i){setTimeout(function(){c.classList.add('revealed');},i*80);});
  }
}

// ── RAF loop (started immediately — before any possible error) ───────────────
function rafLoop(ts){
  requestAnimationFrame(rafLoop);

  // Entrance animation
  if(phase==='animating'&&entranceStart!=null){
    var t=Math.min(1,(ts-entranceStart)/ENTRANCE_DURATION);
    var e=ease(t);
    var zoom=1.12-0.12*e;
    var opVideo=t>0.3?Math.min(1,(t-0.3)/0.5):0;
    document.body.style.transform='scale('+zoom+') translateZ(0)';
    document.body.style.transformOrigin='center center';
    video.style.opacity=opVideo;
    if(t>=1){
      phase='complete';
      document.body.style.transform='';
      document.getElementById('main-header').classList.add('visible');
      document.getElementById('mobile-header').classList.add('visible');
      document.getElementById('main-content').classList.add('visible');
      video.style.opacity='1';
      scrambleEls.forEach(function(s){setTimeout(function(){triggerScramble(s,'in');},s.delay);});
    }
  }

  // Scroll lerp
  var delta=scrollProgress-smoothScrollProgress;
  if(Math.abs(delta)>0.0001)smoothScrollProgress+=delta*0.12;
  else smoothScrollProgress=scrollProgress;
  var p=smoothScrollProgress;

  // Seek video
  if(videoReady&&video.duration&&Math.abs(delta)>0.008){
    seekTo(p*video.duration);
  }

  // Video blur/scale
  var subtleBase=clamp01((p-0.1)/0.45);
  var progressive=clamp01((p-0.55)/0.4);
  var blur=subtleBase*5+progressive*50;
  var scale=1.03+clamp01((p-0.1)/0.9)*0.08;
  video.style.transform='scale('+scale+')';
  video.style.filter='blur('+blur+'px)';

  // Hero parallax
  var heroP=clamp01(p/0.26);
  var hero=document.getElementById('hero');
  hero.style.opacity=1-heroP;
  hero.style.transform='scale('+(1-0.04*heroP)+')';

  // Description parallax
  var descP=clamp01(p/0.12);
  var desc=document.getElementById('hero-description');
  desc.style.opacity=1-descP;
  desc.style.transform='translateY('+(-30*descP)+'px)';

  // Cinematic
  var cin=document.getElementById('cinematic-inner');
  var sy=document.documentElement.scrollTop||document.body.scrollTop;
  var transY=-120*Math.min(1,sy/1000);
  var fracs=[0.08,0.22,0.42,0.65],opacs=[0,1,1,0],cinOp=0;
  if(p<=fracs[0])cinOp=0;
  else if(p<=fracs[1])cinOp=(p-fracs[0])/(fracs[1]-fracs[0]);
  else if(p<=fracs[2])cinOp=1;
  else if(p<=fracs[3])cinOp=1-(p-fracs[2])/(fracs[3]-fracs[2]);
  else cinOp=0;
  cin.style.transform='rotateX(24deg) translateY('+transY+'px) translateZ(15px)';
  cin.style.opacity=cinOp;

  maybeRevealStats();
  animateScramble(ts);

  // Scramble scroll trigger
  if(phase==='complete'){
    if(scrollProgress>0.015){
      scrambleEls.forEach(function(s){if(s.state==='visible'||(s.animDir==='in'&&s.animStart))triggerScramble(s,'out');});
    } else {
      scrambleEls.forEach(function(s){if(s.state==='hidden'||(s.animDir==='out'&&s.animStart))triggerScramble(s,'in');});
    }
  }
}
requestAnimationFrame(rafLoop);

// ── Scroll ───────────────────────────────────────────────────────────────────
function updateScrollProgress(){
  var el=document.documentElement;
  scrollProgress=(el.scrollTop||document.body.scrollTop)/((el.scrollHeight-el.clientHeight)||1);
}
window.addEventListener('scroll',updateScrollProgress,{passive:true});

// ── Lenis (desktop only, graceful failure) ────────────────────────────────────
(function(){
  var isMobile=/Mobi|Android/i.test(navigator.userAgent)||window.innerWidth<768;
  if(!isMobile){
    var s=document.createElement('script');
    s.src='https://unpkg.com/@studio-freight/lenis@1.0.42/dist/lenis.min.js';
    s.onload=function(){
      try{
        var lenis=new Lenis({duration:1.2,easing:function(t){return Math.min(1,1.001-Math.pow(2,-10*t));},smoothWheel:true});
        window.__lenis=lenis;
        lenis.on('scroll',updateScrollProgress);
        (function lenisRaf(ts){lenis.raf(ts);requestAnimationFrame(lenisRaf);})(0);
      }catch(e){}
    };
    document.head.appendChild(s);
  }
})();

    }catch(e){}
  };
  document.head.appendChild(s);
})();

document.getElementById('desktop-hamburger').addEventListener('click',function(){this.classList.toggle('open');});
(function(){
