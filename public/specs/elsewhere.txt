ELSEWHERE — complete build spec

Recreation of a pasted "Step Into Wonder" scroll-experience spec: a single-component React 18 app (portal-archway entrance, mouse-parallax rAF loop, scroll-scrubbed world/portal/curtain zoom, a nine-card arc slider that rotates 80° across the scroll range). Rebranded as Elsewhere, a bespoke travel-and-expedition house — the portal/curtain/floating-destination-arc language already reads as travel, so only the text changed: brand name, hero headline, nav labels, card titles and descriptions, and the two display/body typefaces (Viaoda Libre + Imprima -> Italiana + Jost). All five background/portal/curtain images and the three hero preview photos are the source spec's own asset URLs, used exactly as given — per this repo's rebrand policy (see /CLAUDE.md §1), only the copy changes, not the imagery.

Scroll runway shortened 480vh -> 320vh: every reveal threshold in the file is a fraction of scroll progress (0..1), so the same choreography now completes in noticeably less physical scroll distance — the "fast scroll" standard (/CLAUDE.md §1a), not a mechanics change.

Stack: Vite + React 18 + TypeScript + Tailwind CSS v3 (default config, via postcss).
Dependencies: react@^18.3.1, react-dom@^18.3.1.
Dev dependencies: @types/react@^18.3.12, @types/react-dom@^18.3.1, @vitejs/plugin-react@^4.3.4, autoprefixer@^10.4.20, postcss@^8.4.49, tailwindcss@^3.4.15, typescript@^5.6.3, vite@^5.4.11.

## Setup
1. `npm create vite@latest . -- --template react-ts`
2. Install the exact dependencies listed above.
3. Run `npx tailwindcss init -p` to create tailwind.config.js and postcss.config.js (default config, content globs covering index.html and src/**/*.{js,ts,jsx,tsx}).
4. Replace the generated files below with the exact contents given — every file, verbatim.

function easeInOut(t: number) {
  return t < 0.5 ? 2 * t * t : -1 + (4 - 2 * t) * t
}
function lerp(a: number, b: number, t: number) {
  return a + (b - a) * t
}
function clamp(v: number, min: number, max: number) {
  return Math.min(max, Math.max(min, v))
}

function useIsMobile() {
  const [isMobile, setIsMobile] = useState(() =>
    typeof window !== 'undefined' ? window.innerWidth < 768 : false,
  )
  useEffect(() => {
    const mq = window.matchMedia('(max-width: 767px)')
    const onChange = () => setIsMobile(mq.matches)
    onChange()
    if (mq.addEventListener) mq.addEventListener('change', onChange)
    else mq.addListener(onChange)
    return () => {
      if (mq.removeEventListener) mq.removeEventListener('change', onChange)
      else mq.removeListener(onChange)
    }
  }, [])
  return isMobile
}

function ArcCardSlider({
  cards,
  rotationOffset,
  isMobile,
}: {
  cards: Card[]
  rotationOffset: number
  isMobile: boolean
}) {
  const cardSpacingDeg = isMobile ? 12 : 9
  const centerIndex = 4
  const arcRadius = isMobile ? 700 : 1100
  const cardW = isMobile ? 160 : 220
  const cardH = isMobile ? 175 : 230

  return (
    <div
      style={{
        height: isMobile ? 260 : 360,
        width: '100vw',
        display: 'flex',
        justifyContent: 'center',
        alignItems: 'flex-end',
        overflow: 'visible',
        position: 'relative',
      }}
    >
      {cards.map((card, i) => {
        const baseDeg = (i - centerIndex) * cardSpacingDeg
        const deg = baseDeg - rotationOffset + centerIndex * cardSpacingDeg
        const rad = (deg * Math.PI) / 180
        const x = Math.sin(rad) * arcRadius
        const y = arcRadius - Math.cos(rad) * arcRadius
        const bottom = -y + (isMobile ? 140 : 200)

        return (
          <div
            key={card.title}
            style={{
              position: 'absolute',
              bottom,
              left: `calc(50% + ${x}px - ${cardW / 2}px)`,
              transform: `rotate(${deg}deg)`,
              transformOrigin: `${cardW / 2}px ${arcRadius}px`,
              transition: 'transform 0.05s linear',
              willChange: 'transform',
            }}
          >
            <div
              style={{
                width: cardW,
                height: cardH,
                borderRadius: isMobile ? 18 : 26,
                background: card.color,
                boxShadow: '0 8px 40px rgba(80,40,60,0.18)',
                padding: isMobile ? '12px' : '18px 16px',
                display: 'flex',
                flexDirection: 'column',
                justifyContent: 'space-between',
                overflow: 'hidden',
              }}
            >
              <div style={{ display: 'flex', justifyContent: 'flex-end' }}>
                <div
                  style={{
                    width: 24,
                    height: 24,
                    borderRadius: '50%',
                    border: '1.5px solid rgba(80,50,60,0.3)',
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    fontFamily: 'Jost, sans-serif',
                    fontSize: 10,
                    color: 'rgba(80,50,60,0.6)',
                  }}
                >
                  {String(i + 1).padStart(2, '0')}
                </div>
              </div>
              <div>
                <h3
                  style={{
                    fontFamily: 'Italiana, serif',
                    fontSize: isMobile ? 22 : 30,
                    color: '#3a2530',
                    lineHeight: 1.15,
                    margin: isMobile ? '0 0 4px' : '0 0 8px',

                  }}

                </p>
              </div>
            </div>

}

export default function App() {
  const isMobile = useIsMobile()

  const [scrollProgress, setScrollProgress] = useState(0)
  const scrollRef = useRef(0)
  const containerRef = useRef<HTMLDivElement>(null)

  const [curtainsOpen, setCurtainsOpen] = useState(false)
  const [uiVisible, setUiVisible] = useState(false)
  const [entranceDone, setEntranceDone] = useState(false)

  const worldRef = useRef<HTMLDivElement>(null)
  const cloudsRef = useRef<HTMLDivElement>(null)
  const portalRef = useRef<HTMLDivElement>(null)
  const curtainLRef = useRef<HTMLDivElement>(null)
  const curtainRRef = useRef<HTMLDivElement>(null)

  const raw = useRef({ x: 0, y: 0 })
  const smooth = useRef({ x: 0, y: 0 })

  // Smooth scroll — the scroll-progress effect below reads window.scrollY,
  // which Lenis keeps in sync as it eases, so no downstream changes are
  // needed. window.__lenis is exposed for capture/preview scripts, which
  // must drive it directly (lenis.scrollTo) rather than a bare
  // window.scrollTo, or the eased position fights the capture's own
  // scroll writes.
  useEffect(() => {
    if (matchMedia('(prefers-reduced-motion: reduce)').matches) return
    const lenis = new Lenis({ duration: 1.0, easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)) })
    ;(window as unknown as { __lenis: Lenis }).__lenis = lenis
    let raf = 0
    function tick(time: number) {
      lenis.raf(time)
      raf = requestAnimationFrame(tick)
    }
    raf = requestAnimationFrame(tick)
    return () => {
      cancelAnimationFrame(raf)
      lenis.destroy()
    }
  }, [])

  // entrance sequence
  useEffect(() => {
    const t1 = setTimeout(() => setCurtainsOpen(true), 100)
    const t2 = setTimeout(() => setUiVisible(true), 600)
    const t3 = setTimeout(() => setEntranceDone(true), 2200)
    return () => {
      clearTimeout(t1)
      clearTimeout(t2)
      clearTimeout(t3)
    }
  }, [])

  // scroll progress
  useEffect(() => {
    const onScroll = () => {
      const el = containerRef.current
      if (!el) return
      const max = el.scrollHeight - window.innerHeight
      const p = clamp(max > 0 ? window.scrollY / max : 0, 0, 1)
      scrollRef.current = p
      setScrollProgress(p)
    }
    onScroll()
    window.addEventListener('scroll', onScroll, { passive: true })
    return () => window.removeEventListener('scroll', onScroll)
  }, [])

  // mouse parallax
  useEffect(() => {
    const onMove = (e: MouseEvent) => {
      raw.current.x = (e.clientX / window.innerWidth - 0.5) * 2
      raw.current.y = (e.clientY / window.innerHeight - 0.5) * 2
    }
    window.addEventListener('mousemove', onMove, { passive: true })
    return () => window.removeEventListener('mousemove', onMove)
  }, [])

  // rAF loop driving layer transforms
  useEffect(() => {
    let raf = 0
    const tick = () => {
      smooth.current.x = lerp(smooth.current.x, raw.current.x, 0.07)
      smooth.current.y = lerp(smooth.current.y, raw.current.y, 0.07)
      const rx = -smooth.current.x
      const ry = -smooth.current.y
      const ep = easeInOut(scrollRef.current)

      if (worldRef.current) {
        const s = lerp(1, 1.18, ep)
        worldRef.current.style.transform = `scale(${s}) translate(${rx * 6}px, ${ry * 6}px)`
        worldRef.current.style.transformOrigin = '50% 50%'
      }
      if (cloudsRef.current) {
        const s = lerp(1, 1.4, ep)
        cloudsRef.current.style.transform = `scale(${s}) translate(${rx * 9}px, ${ry * 9 * 0.4}px)`
        cloudsRef.current.style.transformOrigin = '50% 100%'
      }
      if (portalRef.current) {
        const s = lerp(1, 7.5, ep)
        portalRef.current.style.transform = `scale(${s}) translate(${rx * 7}px, ${ry * 7}px)`
        portalRef.current.style.transformOrigin = '52% 38%'
      }
      const curtainTransition = entranceDone ? 'none' : 'transform 1.8s cubic-bezier(0.16, 1, 0.3, 1)'
      if (curtainLRef.current) {
        const totalShift = (curtainsOpen ? 62 : 0) + lerp(0, 150, ep)
        const s = lerp(1, 1.3, ep)
        curtainLRef.current.style.transform = `translateX(calc(-${totalShift}% + ${rx * 14}px)) translateY(${ry * 14 * 0.3}px) scale(${s})`
        curtainLRef.current.style.transformOrigin = 'left center'
        curtainLRef.current.style.transition = curtainTransition
      }
      if (curtainRRef.current) {
        const totalShift = (curtainsOpen ? 62 : 0) + lerp(0, 150, ep)
        const s = lerp(1, 1.3, ep)
        curtainRRef.current.style.transform = `translateX(calc(${totalShift}% + ${rx * 14}px)) translateY(${ry * 14 * 0.3}px) scale(${s})`
        curtainRRef.current.style.transformOrigin = 'right center'
        curtainRRef.current.style.transition = curtainTransition
      }

      raf = requestAnimationFrame(tick)
    }
    raf = requestAnimationFrame(tick)
    return () => cancelAnimationFrame(raf)
  }, [curtainsOpen, entranceDone])

  const p = scrollProgress
  const portalOpacity = p < 0.65 ? 1 : clamp(1 - (p - 0.65) / 0.2, 0, 1)

      </>
    ),
    [],
  )

  return (
    <div ref={containerRef} style={{ height: `${RUNWAY_VH}vh`, position: 'relative' }}>
