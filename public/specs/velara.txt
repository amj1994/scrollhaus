Velara — complete build spec

Recreation of a pasted 'AI Developer Solver Hero Velara' component spec, with new original branding kept as-is (Velaraai/DevSolver were the source's own generic placeholder names) but the hosted hero-video background — hotlinked from another site's CDN — replaced with a live canvas particle-stream effect in the same violet palette, rendering the same downward falling-light motion without reusing anyone else's footage.

Stack: Vite + React 19 + TypeScript + Tailwind CSS v4 (via @tailwindcss/vite).
Dependencies: framer-motion@^12.0.0, lucide-react@^0.475.0, react@^19.0.0, react-dom@^19.0.0.
Dev dependencies: @tailwindcss/vite@^4.2.1, @types/react@^19.0.10, @types/react-dom@^19.0.4, @vitejs/plugin-react@^4.3.4, tailwindcss@^4.2.1, typescript@^5.8.0, vite@^7.0.0.

## Setup
1. `npm create vite@latest . -- --template react-ts`
2. Install the exact dependencies listed above (`npm install framer-motion@12.0.0 lucide-react@0.475.0 react@19.0.0 react-dom@19.0.0` plus dev deps `npm install -D @tailwindcss/vite@4.2.1 @types/react@19.0.10 @types/react-dom@19.0.4 @vitejs/plugin-react@4.3.4 tailwindcss@4.2.1 typescript@5.8.0 vite@7.0.0`).
3. Add a `"@/*": ["./src/*"]` path alias in tsconfig.json and vite.config.ts (resolve.alias), and add the `@tailwindcss/vite` plugin to vite.config.ts.
4. Replace the generated files below with the exact contents given — every file, verbatim.

function FallingStreams() {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    const reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    const colors = ["168,84,247", "146,51,234", "192,132,252"];
    let raf = 0;
    let w = 0;
    let h = 0;
    let drops: { x: number; y: number; len: number; speed: number; c: string; alpha: number }[] = [];

    const resize = () => {
      w = canvas.clientWidth;
      h = canvas.clientHeight;
      canvas.width = w * devicePixelRatio;
      canvas.height = h * devicePixelRatio;
      ctx.setTransform(1, 0, 0, 1, 0, 0);
      ctx.scale(devicePixelRatio, devicePixelRatio);
      const count = Math.round(w / 11);
      drops = Array.from({ length: count }, () => ({
        x: Math.random() * w,
        y: Math.random() * h - h,
        len: 80 + Math.random() * 180,
        speed: 1.4 + Math.random() * 3,
        c: colors[Math.floor(Math.random() * colors.length)],
        alpha: 0.22 + Math.random() * 0.38,
      }));
    };

    const draw = () => {
      ctx.clearRect(0, 0, w, h);
      for (const d of drops) {
        d.y += d.speed;
        if (d.y - d.len > h) {
          d.y = -Math.random() * h * 0.3;
          d.x = Math.random() * w;
        }
        const grad = ctx.createLinearGradient(d.x, d.y - d.len, d.x, d.y);
        grad.addColorStop(0, `rgba(${d.c},0)`);
        grad.addColorStop(1, `rgba(${d.c},${d.alpha})`);
        ctx.strokeStyle = grad;
        ctx.lineWidth = 1.4;
        ctx.beginPath();
        ctx.moveTo(d.x, d.y - d.len);
        ctx.lineTo(d.x, d.y);
        ctx.stroke();

        ctx.shadowColor = `rgba(${d.c},0.8)`;
        ctx.shadowBlur = 6;
        ctx.fillStyle = `rgba(${d.c},${Math.min(1, d.alpha + 0.45)})`;
        ctx.beginPath();
        ctx.arc(d.x, d.y, 1.6, 0, Math.PI * 2);
        ctx.fill();
        ctx.shadowBlur = 0;
      }
      if (!reduced) raf = requestAnimationFrame(draw);
    };

    resize();
    draw();
    window.addEventListener("resize", resize);
    return () => {
      window.removeEventListener("resize", resize);
      cancelAnimationFrame(raf);
    };
  }, []);

  return <canvas ref={canvasRef} className="w-full h-full" />;
}
