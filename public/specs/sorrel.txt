SORREL — Hero Section — complete build spec

A finished, single-file-hero build supplied ready-made: a full-bleed water-brand
hero with a self-hosted macro water/moss video background, tracked-out mono
wordmark, a two-layer gutter/frame spacing system, and a custom teardrop/leaf
inline-SVG logo mark. Clamp-scaled display type ("TASTE" / "THE" / "QUIET")
positioned at three corners around a hairline frame.

Video: originally streamed from Gumlet at its raw capture bitrate (5.6Mbps,
1080p, with an unused silent audio track). Re-encoded (audio stripped, crf 27)
and self-hosted at assets/video/sorrel-hero.mp4 instead — same visual quality,
roughly a fifth of the bytes, and no dependency on an external account staying
active.

Motion: Lenis smooth scroll wired in (the page itself is a single fixed
viewport, so there's nothing to scroll, but every showcase in this library
ships Lenis by default).

Stack: Vite + React 18 + TypeScript + Tailwind CSS.

## Setup
`npm install`, then `npm run build`. Serve the built `dist/` (or `npm run dev`
for local iteration). The hero video must be present at
`assets/video/sorrel-hero.mp4` relative to the served root.

## Files

### src/App.tsx
```tsx
import { useEffect } from 'react'
import Lenis from 'lenis'
import Hero from './components/Hero'

function useLenis() {
  useEffect(() => {
    if (matchMedia('(prefers-reduced-motion: reduce)').matches) return
    const lenis = new Lenis({ duration: 1.0, easing: (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)) })
    ;(window as unknown as { __lenis: Lenis }).__lenis = lenis
    let raf = 0
    function tick(time: number) {
      lenis.raf(time)
      raf = requestAnimationFrame(tick)
    }
    raf = requestAnimationFrame(tick)
    return () => {
      cancelAnimationFrame(raf)
      lenis.destroy()
    }
  }, [])
}

export default function App() {
  useLenis()
  return <Hero />
}
```

// --- Mobile Menu ---
const MOBILE_LINKS = [...NAV_LINKS, 'See map']

function MobileMenu({
  open,
  onClose,
}: {
  open: boolean
  onClose: () => void
}) {
  useEffect(() => {
    if (open) {
      document.body.style.overflow = 'hidden'
    } else {
      document.body.style.overflow = ''
    }
    return () => {
      document.body.style.overflow = ''
    }
  }, [open])

  return (
    <div
      className="absolute inset-0 z-50 flex flex-col transition-all duration-500 ease-[cubic-bezier(0.76,0,0.24,1)]"
      style={{
        opacity: open ? 1 : 0,
        pointerEvents: open ? 'auto' : 'none',
      }}
    >
      {/* Scrim */}
      <div
        className="absolute inset-0 bg-black/90 backdrop-blur-md"
        onClick={onClose}
      />

      {/* Close button */}
      <button
        aria-label="Close menu"
        onClick={onClose}
        className="absolute right-4 top-4 z-10 flex h-12 w-12 items-center justify-center rounded-full ring-1 ring-white/20 transition-all duration-[400ms] ease-[cubic-bezier(0.76,0,0.24,1)] sm:right-6 sm:top-6"
        style={{

        }}
      >
        <svg
          width="18"
          height="18"
          viewBox="0 0 18 18"
          fill="none"
          stroke="currentColor"
          strokeWidth="1.5"
          strokeLinecap="round"
          className="text-white"
        >
          <path d="M1 1L17 17M17 1L1 17" />
        </svg>
      </button>

      {/* Link stack */}
      <div className="relative z-10 flex flex-1 flex-col items-center justify-center gap-1">
        {MOBILE_LINKS.map((link, i) => (
          <a
            key={link}
            href="#"
            onClick={onClose}

            style={{
              opacity: open ? 1 : 0,

        className="relative z-10 flex items-center gap-3 px-6 pb-8"

          opacity: open ? 1 : 0,
          transform: open ? 'translateY(0)' : 'translateY(-24px)',

        </span>

    </div>
  )
}

// --- Hero ---
export default function Hero() {
  const videoRef = useRef<HTMLVideoElement>(null)
  const [showLabels, setShowLabels] = useState(false)
  const [menuOpen, setMenuOpen] = useState(false)

  useEffect(() => {
    const video = videoRef.current
    if (!video) return

    const handleTimeUpdate = () => {
      if (!video.duration) return
      setShowLabels(video.currentTime / video.duration >= 0.75)
    }

    video.addEventListener('timeupdate', handleTimeUpdate)
    return () => video.removeEventListener('timeupdate', handleTimeUpdate)
  }, [])

  // Corner bracket helper
  const cornerBracket = (position: string) => {
    const borders: Record<string, string> = {
      'top-left': 'border-l-2 border-t-2 top-0 left-0',
      'top-right': 'border-r-2 border-t-2 top-0 right-0',
      'bottom-left': 'border-l-2 border-b-2 bottom-0 left-0',
      'bottom-right': 'border-r-2 border-b-2 bottom-0 right-0',
    }
    return (
      <span
        key={position}
        className={`absolute h-[10px] w-[10px] border-white sm:h-[14px] sm:w-[14px] ${borders[position]}`}
      />
    )
  }

  return (
    <section className="relative h-[100dvh] w-full overflow-hidden bg-[#010101] font-mono">
      {/* Background video */}
      <video
        ref={videoRef}
        src={GUMLET_VIDEO}
        className="absolute inset-0 h-full w-full object-cover"
        autoPlay
        loop
        muted
        playsInline
      />

      {/* Layer A — outer gutter */}
      <div className="relative z-10 mx-auto flex h-full flex-col p-2 sm:p-3 md:p-4 lg:p-5">
        {/* Frame */}
        <div className="relative flex flex-1 flex-col border border-white/10">
          {/* Corner brackets */}
          {['top-left', 'top-right', 'bottom-left', 'bottom-right'].map(
            cornerBracket,
          )}
          {/* Centre ticks */}
          <span className="absolute -top-[1px] left-1/2 h-px w-[12px] -translate-x-1/2 bg-white" />
          <span className="absolute -bottom-[1px] left-1/2 h-px w-[12px] -translate-x-1/2 bg-white" />

          {/* Nav */}
          <Nav menuOpen={menuOpen} setMenuOpen={setMenuOpen} />

          {/* Mobile menu (always mounted) */}
          <MobileMenu open={menuOpen} onClose={() => setMenuOpen(false)} />

          {/* Layer B — content padding */}
          <div className="relative flex flex-1 flex-col justify-between p-3 sm:p-4 md:p-6 lg:p-8">
            {/* Top row */}
            <div className="flex w-full items-start justify-between">
              <span
                className="text-[clamp(2rem,8.5vw,9.5rem)] font-normal uppercase leading-[0.9] text-white"
              >
                TASTE
              </span>
              <span
                className="text-[clamp(2rem,8.5vw,9.5rem)] font-normal uppercase leading-[0.9] text-white"
              >
                THE
              </span>
            </div>

            {/* Bottom row */}
            <div className="flex w-full items-end justify-between">
              {/* Description block */}
              <div className="max-w-[140px] sm:max-w-[196px]">
                <AtomDecor className="h-10 w-10 sm:h-12 sm:w-12 md:h-16 md:w-16 lg:h-20 lg:w-20" />
                <p className="mt-2 text-[10px] uppercase leading-[1.3] tracking-wide text-white sm:mt-3 sm:text-xs md:text-sm lg:text-base xl:text-xl xl:leading-[1.2]">
                  Drawn from rock, bottled unhurried, for the part of the day
                  that belongs to you
                </p>
              </div>
              <span
                className="text-[clamp(2rem,8.5vw,9.5rem)] font-normal uppercase leading-[0.9] text-white"
              >
                QUIET
              </span>
            </div>

            {/* Video-driven callout labels */}
            <div
              className="pointer-events-none absolute inset-0 p-3 transition-opacity duration-700 ease-in-out sm:p-4 md:p-6 lg:p-8"
              style={{ opacity: showLabels ? 1 : 0 }}
            >
              {/* Spring-Drawn — top-right area */}
              <div className="absolute right-[12%] top-[32%] sm:right-[22%] sm:top-[35%] md:right-[25%] lg:right-[28%]">
                <DashedLabel text="Spring-Drawn" variant="top-right" />
              </div>

              {/* Slow-Filtered — bottom-left area */}
              <div className="absolute bottom-[32%] left-[5%] sm:bottom-[35%] sm:left-[12%] md:left-[16%] lg:left-[18%]">
                <DashedLabel text="Slow-Filtered" variant="bottom-left" />
              </div>
            </div>
          </div>
        </div>
      </div>
    </section>
  )
}
