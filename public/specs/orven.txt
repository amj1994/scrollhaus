Orven — complete build spec

Recreation of a pasted 'Orven' luxury performance-eyewear scroll-hero spec: a vanilla HTML/CSS/JS single-page experience with two video-scrub phases, a crossfade, feature panels driven by position:relative+top for mix-blend-mode-safe motion, an RAF-lerp video scrub engine, and a word/char stagger reveal system. The spec's own two hero video URLs pointed to another party's Higgsfield account, so both are original Higgsfield-generated loops instead: a rotating product shot of performance sunglasses on a reflective surface, and a macro push-in on a lens edge, encoded with a dense keyframe interval since these are scroll-scrubbed (seeked), not just played.

Stack: plain static HTML/CSS/vanilla JS. No build step, no framework.

## Setup
Create one file named index.html with exactly this content, and serve it as a static file (any static server, e.g. `python3 -m http.server`).

<script>
(function () {
  // ---------- Dev grid overlay ("G" key) ----------
  const devGrid = document.getElementById('devGrid');
  document.addEventListener('keydown', (e) => {
    if (e.key === 'g' || e.key === 'G') devGrid.classList.toggle('visible');
  });

  // ---------- Split-reveal ----------
  function splitReveal(el, mode, stagger) {
    const text = el.textContent;
    el.textContent = '';
    const units = mode === 'char' ? [...text] : text.split(' ');
    units.forEach((unit, i) => {
      const span = document.createElement('span');
      span.className = 'reveal-word';
      span.textContent = unit;
      span.style.setProperty('--i', i);
      span.style.setProperty('--stagger', stagger + 's');
      el.appendChild(span);
      if (mode === 'word' && i < units.length - 1) {
        el.appendChild(document.createTextNode(' '));
      }
    });
  }

  splitReveal(document.getElementById('heroTitle'), 'char', 0.025);
  splitReveal(document.getElementById('heroCaption'), 'word', 0.05);
  splitReveal(document.getElementById('heroInfoText'), 'word', 0.05);
  splitReveal(document.getElementById('rightPanelText'), 'word', 0.05);
  splitReveal(document.getElementById('leftPanelText'), 'word', 0.05);

  // Hero title cluster reveals immediately on page load.
  const heroUi = document.getElementById('heroUi');
  requestAnimationFrame(() => heroUi.classList.add('reveal-active'));

  // ---------- Video scrub engine (RAF-lerp) ----------
  function makeScrubber(video) {
    let target = 0;
    let current = 0;
    let lastSeek = 0;
    let raf = null;
    let primed = false;

    video.addEventListener('loadedmetadata', () => {
      if (primed) return;
      primed = true;
      const p = video.play();
      if (p && p.then) p.then(() => video.pause()).catch(() => {});
      else video.pause();
    });

    function tick(now) {
      current += (target - current) * 0.09;
      if (Math.abs(target - current) < 0.002) current = target;

      const drift = Math.abs(video.currentTime - current);
      if (drift > 0.02 && now - lastSeek > 30) {
        try { video.currentTime = current; } catch (e) {}
        lastSeek = now;
      }

      if (Math.abs(target - current) > 0.001) {
        raf = requestAnimationFrame(tick);
      } else {
        raf = null;
      }
    }

    return {
      setTarget(t) {
        const dur = video.duration || 0;
        target = Math.max(0, Math.min(dur, t));
        if (!raf) raf = requestAnimationFrame(tick);
      },
    };
  }

  const video1 = document.getElementById('video1');
  const video2 = document.getElementById('video2');
  const scrub1 = makeScrubber(video1);
  const scrub2 = makeScrubber(video2);

  // ---------- Scroll-driven timeline ----------
  const SCRUB1_VH = 3;
  const FADE_VH = 3;
  const SCRUB2_VH = 5;
  const TOTAL_VH = SCRUB1_VH + FADE_VH + SCRUB2_VH + 1;

  const heroEl = document.querySelector('.hero');
  const heroCaption = document.getElementById('heroCaption');
  const panelRight = document.getElementById('panelRight');
  const panelLeft = document.getElementById('panelLeft');

  function clamp(v, a, b) { return Math.max(a, Math.min(b, v)); }

  function panelMotion(panel, subProgress) {
    const vh = window.innerHeight;
    const h = panel.offsetHeight || 0;
    const start = vh + 60;
    const travel = vh + h + 60;
    panel.style.top = (start - subProgress * travel) + 'px';
  }

  function applyReveal(el, subProgress) {
    if (subProgress < 0.12) {
      el.classList.remove('reveal-active', 'reveal-exit');
    } else if (subProgress <= 0.88) {
      el.classList.add('reveal-active');
      el.classList.remove('reveal-exit');
    } else {
      el.classList.add('reveal-exit');
      el.classList.remove('reveal-active');
    }
  }

  function update() {
    const vh = window.innerHeight;
    const start = heroEl.offsetTop;
    const scrollVh = clamp((window.scrollY - start) / vh, 0, TOTAL_VH);

    const p1 = clamp(scrollVh / SCRUB1_VH, 0, 1);
    const p2 = clamp((scrollVh - SCRUB1_VH) / FADE_VH, 0, 1);
    const p3 = clamp((scrollVh - SCRUB1_VH - FADE_VH) / SCRUB2_VH, 0, 1);

    // Phase 1 — video 1 scrub + hero UI fade/dissolve.
    if (video1.duration) scrub1.setTarget(p1 * video1.duration);

    const uiFade = clamp((p1 - 0.8) / 0.2, 0, 1);
    heroUi.style.opacity = 1 - uiFade;
    heroUi.style.transform = 'translateY(' + (-120 * uiFade) + 'px)';
    if (uiFade > 0.15) {
      heroUi.classList.add('reveal-exit');
      heroUi.classList.remove('reveal-active');
    } else {
      heroUi.classList.add('reveal-active');
      heroUi.classList.remove('reveal-exit');
    }

    // Phase 2 — crossfade.
    video1.style.opacity = 1 - p2;
    video1.style.filter = 'blur(' + (p2 * 80) + 'px)';
    video2.style.opacity = p2;

    // Phase 3 — video 2 scrub + feature panels.
    const scrub2Progress = clamp(p3 / 0.8, 0, 1);
    if (video2.duration) scrub2.setTarget(scrub2Progress * video2.duration);

    const rightSub = clamp((p3 - 0) / (0.4 - 0), 0, 1);
    const leftSub = clamp((p3 - 0.45) / (0.85 - 0.45), 0, 1);

    panelMotion(panelRight, rightSub);
    panelMotion(panelLeft, leftSub);
    applyReveal(document.getElementById('rightPanelText'), rightSub);
    applyReveal(document.getElementById('leftPanelText'), leftSub);
  }

  window.addEventListener('scroll', update, { passive: true });
  window.addEventListener('resize', update);
  update();
})();
</script>
</body>
</html>
