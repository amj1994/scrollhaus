Frostbound — complete build spec

Recreation of a pasted 'Frostbound — Wander to Greenland' single-file hero spec (a snow-covered Icelandic-fjord travel hero with a headline sandwiched between two stacked copies of the same photo, plus a clip-path mobile overlay menu). The spec's own image URLs pointed to another party's asset hosting, so both photo layers here are an original Higgsfield-generated Icelandic-fjord photograph instead: the background layer is the plain photo, and the foreground layer is the same photograph with the sky cut to transparent (via a custom Python script, feathered along the real ridge silhouette) so the WANDER headline reads from behind the mountains and in front of the sky, exactly as the spec describes.

Stack: plain static HTML/CSS/vanilla JS. No build step, no framework.

## Setup
Create one file named index.html with exactly this content, and serve it as a static file (any static server, e.g. `python3 -m http.server`).

<script>
  (function () {
    var body = document.body;
    var burger = document.querySelector('.burger');
    var menu = document.getElementById('menu');

    function setOpen(open) {
      body.classList.toggle('nav-open', open);
      burger.setAttribute('aria-expanded', String(open));
      burger.setAttribute('aria-label', open ? 'Close menu' : 'Open menu');
      menu.setAttribute('aria-hidden', String(!open));
    }

    burger.addEventListener('click', function () {
      setOpen(!body.classList.contains('nav-open'));
    });

    menu.querySelectorAll('a').forEach(function (a) {
      a.addEventListener('click', function () { setOpen(false); });
    });

    document.addEventListener('keydown', function (e) {
      if (e.key === 'Escape' && body.classList.contains('nav-open')) {
        setOpen(false);
        burger.focus();
      }
    });

    var mq = window.matchMedia('(min-width: 901px)');
    function handleMqChange(e) {
      if (e.matches) setOpen(false);
    }
    if (mq.addEventListener) mq.addEventListener('change', handleMqChange);
    else if (mq.addListener) mq.addListener(handleMqChange);
  })();

  (function () {
    var hero = document.querySelector('.hero');
    hero.addEventListener('animationend', function (e) {
      if (e.target.closest('.menu')) return;
      e.target.classList.add('is-revealed');
    });
  })();
</script>
