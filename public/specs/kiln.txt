KILN — complete build spec

Recreation of a pasted "Ascend" web3-marketing-agency spec: a pixel-exact scaled (--u custom property) landing page with a fixed, full-viewport background video scrubbed by scroll position via native <video> currentTime seeking (all-intra H.264 for instant arbitrary seeks), a scripted entrance choreography (masked headline reveal, staggered header lift-ins), scroll-triggered section reveals, and three responsive regimes (desktop scale system, tablet grid reflow, mobile stack) sharing one collapsed-nav component. Rebranded top to bottom as KILN, a bespoke fragrance atelier — the source spec's agency-serves-clients structure maps directly onto a scent studio that crafts fragrance FOR other brands, so the DOM shape (About / Clients / Contact nav, a client-logo grid) needed no restructuring, only new copy, palette (amber/black replacing the source's cool blue-grey with a single orange accent), and hero footage. The hero background is an original Higgsfield generation — a continuous glass-blowing-atelier shot (an arch, a rising tiered tower, a sweep past the lens, a final glass column with flame-like amber gradients) — replacing the source spec's hardcoded CloudFront demo clips.

Stack: plain static HTML/CSS/vanilla JS. No build step, no framework.

## Setup
Create one file named index.html with exactly this content, plus an assets/hero-poster.jpg file (the video's own frame 0, so the poster-to-video handover is invisible), and serve it as a static file (any static server, e.g. `python3 -m http.server`).

/* Smooth scroll — the scrub loop below reads window.pageYOffset each
   tick, which Lenis keeps in sync as it eases, so no downstream changes
   are needed. window.__lenis is exposed for capture scripts, which must
   drive it directly (lenis.scrollTo) rather than a bare window.scrollTo. */
if (!matchMedia('(prefers-reduced-motion: reduce)').matches) {
  var lenis = new Lenis({ duration: 1.0, easing: function (t) { return Math.min(1, 1.001 - Math.pow(2, -10 * t)); } });
  window.__lenis = lenis;
  requestAnimationFrame(function raf(time) { lenis.raf(time); requestAnimationFrame(raf); });
}

/* ── entrance sequence ─────────────────────────────────────── */
(function () {
  var root = document.documentElement;
  try {
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
  } catch (e) { return; }
  root.classList.add('anim');

  var TOTAL = 1650;
  var shown = false;
  function finish() {
    if (!root.classList.contains('anim-run') && !root.classList.contains('anim')) return;
    root.classList.remove('anim-run');
    root.classList.remove('anim');
  }
  function start() {
    if (shown) return;
    shown = true;
    root.classList.add('anim-run');
    setTimeout(finish, TOTAL);
  }
  function begin() {
    var poster = document.getElementById('poster');
    var fontsReady = (document.fonts && document.fonts.ready) ? document.fonts.ready : Promise.resolve();
    var posterReady = (poster && poster.decode) ? poster.decode().catch(function () {}) : Promise.resolve();
    var timeout = new Promise(function (res) { setTimeout(res, 1200); });
    Promise.race([Promise.all([fontsReady, posterReady]), timeout]).then(function () {
      requestAnimationFrame(start);
    });
  }
  if (document.visibilityState === 'hidden') {
    var onVis = function () {
      if (document.visibilityState !== 'hidden') {
        document.removeEventListener('visibilitychange', onVis);
        begin();
      }
    };
    document.addEventListener('visibilitychange', onVis);
  } else {
    begin();
  }
})();

/* ── the scroll-scrubbed scene ─────────────────────────────── */
(function () {
  var scene = document.getElementById('scene');
  var v = document.getElementById('scene-video');
  if (!scene || !v) return;

  var SRC_HD = 'assets/hero-hd.mp4';
  var SRC_SD = 'assets/hero-sd.mp4';

  var px = Math.max(window.innerWidth, window.innerHeight) * Math.min(window.devicePixelRatio || 1, 2);
  var saveData = !!(navigator.connection && navigator.connection.saveData);
  v.src = (px > 1500 && !saveData) ? SRC_HD : SRC_SD;

  var KEYS = [['top', 0, 50], ['about', .56, 78], ['clients', .86, 46]];
  var TAU = 120;

  var stops = [];
  var max = 1;
  function measure() {
    max = Math.max(1, document.documentElement.scrollHeight - window.innerHeight);
    stops = KEYS.map(function (k) {
      var el = document.getElementById(k[0]);
      var top = el ? el.getBoundingClientRect().top + window.pageYOffset : 0;
      return [Math.min(top, max), k[1]];
    });
    stops.push([max, 1]);
  }

  function progress(y) {
    if (!stops.length) return 0;
    if (y <= stops[0][0]) return stops[0][1];
    for (var i = 1; i < stops.length; i++) {
      if (y <= stops[i][0]) {
        var a = stops[i - 1], b = stops[i];
        var t = b[0] === a[0] ? 0 : (y - a[0]) / (b[0] - a[0]);
        return a[1] + (b[1] - a[1]) * t;
      }
    }
    return 1;
  }

  function pan(p) {
    if (p <= KEYS[0][1]) return KEYS[0][2];
    for (var i = 1; i < KEYS.length; i++) {
      if (p <= KEYS[i][1]) {
        var a = KEYS[i - 1], b = KEYS[i];
        var t = b[1] === a[1] ? 0 : (p - a[1]) / (b[1] - a[1]);
        return a[2] + (b[2] - a[2]) * t;
      }
    }
    return KEYS[KEYS.length - 1][2];
  }

  var panX = -1, veilO = -1;
  function orient() {
    var portrait = window.innerWidth / window.innerHeight < 1;
    if (!portrait) {
      v.style.objectPosition = '';
      veil.style.opacity = '';
      panX = -1; veilO = -1;
    }
    return portrait;
  }

  var veil = document.getElementById('veil');
  var dur = 0, cur = 0, target = 0, ready = false, last = 0, queued = false, reduce = false;
  try { reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches; } catch (e) {}

  function kick() {
    if (queued) return;
    queued = true;
    requestAnimationFrame(tick);
  }

  function tick(now) {
    queued = false;
    var dt = last ? Math.min(64, now - last) : 16;
    last = now;

    target = progress(window.pageYOffset) * dur;
    if (reduce) {
      cur = target;
    } else {
      cur += (target - cur) * (1 - Math.exp(-dt / TAU));
      if (Math.abs(target - cur) < 1 / 300) cur = target;
    }

    var off = Math.abs(v.currentTime - cur) > 1 / 90;
    if (ready && off && !v.seeking) {
      try { v.currentTime = cur; } catch (e) {}
    }

    var portrait = orient();
    if (portrait && dur) {
      var p = cur / dur;
      var x = pan(p);
      var o = Math.max(0, Math.min(1, (p - .18) / .3)) * .5;
      if (Math.abs(x - panX) > .05) { panX = x; v.style.objectPosition = x + '% 50%'; }
      if (Math.abs(o - veilO) > .004) { veilO = o; veil.style.opacity = o; }
    }

    if (cur !== target || off || v.seeking) kick();
    else last = 0;
  }

  v.addEventListener('loadedmetadata', function () {
    dur = Math.max(0, v.duration - 1 / 30);
    measure();
    cur = target = progress(window.pageYOffset) * dur;
    ready = true;
    try { v.currentTime = cur; } catch (e) {}
  });

  v.addEventListener('seeked', function () {
    scene.classList.add('is-ready');
    kick();
  });

  function prime() {
    var p = v.play();
    if (p && p.then) {
      p.then(function () { v.pause(); v.currentTime = cur; }).catch(function () {});
    }
  }
  v.addEventListener('loadeddata', prime, { once: true });
  window.addEventListener('touchstart', prime, { once: true, passive: true });

  window.addEventListener('scroll', kick, { passive: true });
  window.addEventListener('resize', function () { orient(); measure(); kick(); });
  if ('ResizeObserver' in window) {
    new ResizeObserver(function () { measure(); kick(); }).observe(document.body);
  }
  if (document.fonts && document.fonts.ready) document.fonts.ready.then(function () { measure(); kick(); });

  orient();
  measure();
})();
</script>
</body>
</html>
